// 由自然语言生成的流程扩展代码
export async function transform(interview, context) {
  const transcript = await tools.transcribe({
    media: interview.file,
    diarization: true,
    dialectBoost: context.region
  });

  const draft = await agents.run('写易 · 智能写作', {
    transcript,
    structure: '现场-人物-数据-政策',
    citations: 'required'
  });

  return await workflow.humanReview(draft, {
    blockOn: ['P1'],
    aigcLabel: true
  });
}
